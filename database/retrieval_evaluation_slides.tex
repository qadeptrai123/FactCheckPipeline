\begin{frame}{Pipeline truy hồi}
  \begin{enumerate}
    \item Từ output refine, tạo gói truy vấn gồm:
    \begin{itemize}
      \item \texttt{text\_queries}: claim đã chuẩn hoá, semantic queries, claim atoms, verification targets;
      \item \texttt{keyword\_query}: keywords và verification targets;
      \item \texttt{visual\_queries}: visual queries và visible evidence.
    \end{itemize}
    \item Truy hồi trong Qdrant theo ba nhánh:
    \begin{itemize}
      \item text dense với Vietnamese bi-encoder;
      \item text sparse với sparse vector;
      \item image lane với CLIP hoặc CLIP fine-tuned.
    \end{itemize}
    \item Hợp nhất ứng viên bằng weighted Reciprocal Rank Fusion.
    \item Nếu bật reranker, dùng ViRanker để tăng điểm các ứng viên text.
    \item Tách kết quả thành hai lane: text và image, rồi lấy top-k.
  \end{enumerate}
\end{frame}

\begin{frame}{Cách xác định một hit}
  \begin{block}{Text/source hit}
    Một retrieved text được xem là đúng nguồn nếu URL sau chuẩn hoá khớp với \texttt{text\_evidences\_url}.
  \end{block}

  \begin{block}{Text evidence hit}
    Nếu URL không khớp, hệ thống so khớp token giữa retrieved text và gold text evidence. Hit khi:
    \[
      \mathrm{coverage} \ge 0.60
      \quad \text{hoặc} \quad
      F_1 \ge 0.45
    \]
  \end{block}

  \begin{block}{Image evidence hit}
    Retrieved image được xem là đúng khi \texttt{image\_path} sau chuẩn hoá khớp với \texttt{image\_evidence\_path}.
  \end{block}
\end{frame}

\begin{frame}{Metric đánh giá}
  Với mỗi claim, metric được tính trên top-k của text lane và image lane, sau đó lấy trung bình trên toàn bộ claim.

  \vspace{0.2cm}
  \begin{itemize}
    \item \textbf{Source Hit@k}: có ít nhất một URL đúng trong top-k của text hoặc image lane.
    \item \textbf{Evidence Hit@k}: có ít nhất một text evidence hit hoặc image exact hit trong top-k.
    \item \textbf{MRR@3}: nghịch đảo rank của hit đầu tiên trong top-3.
  \end{itemize}

  \vspace{0.2cm}
  \[
    \hitk{k} = \frac{1}{N}\sum_{i=1}^{N}\mathbb{1}[\exists\ \mathrm{hit}\ \mathrm{trong}\ \mathrm{top}\ k]
  \]
  \[
    \mrr = \frac{1}{N}\sum_{i=1}^{N}
    \begin{cases}
      1/r_i, & r_i \le 3 \\
      0, & \text{không có hit trong top-3}
    \end{cases}
  \]
\end{frame}

\end{frame}

\begin{frame}{LLM-as-a-judge cho explanation}
  \begin{itemize}
    \item Evaluator: GPT-4o mini.
    \item Input: claim, gold label, model verdict, gold reason, generated explanation.
    \item Mỗi explanation được chấm 5 tiêu chí trên thang 1--5:
    \begin{itemize}
      \item factual consistency;
      \item verdict alignment;
      \item completeness;
      \item Vietnamese clarity;
      \item no hallucination.
    \end{itemize}
    \item Nhãn tổng hợp:
    \begin{itemize}
      \item \texttt{pass}: overall score $\ge 4.0$ và không tiêu chí nào dưới 3;
      \item \texttt{borderline}: overall score $\ge 3.0$ nhưng chưa đạt pass;
      \item \texttt{fail}: overall score $< 3.0$ hoặc có mâu thuẫn nghiêm trọng.
    \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{Nhận xét chính}
  \begin{itemize}
    \item Cấu hình tốt nhất theo \texttt{source\_hit\_at\_3}: Gemini 2.5 Flash + semantic collection + CLIP + reranker, đạt \textbf{0.833}.
    \item Cấu hình tốt nhất theo \texttt{evidence\_hit\_at\_3}: GPT-4o mini + semantic collection + CLIP/CLIP-FT + reranker, đạt \textbf{0.922}.
    \item Reranker cải thiện rõ rệt nhóm cấu hình đứng đầu: các kết quả top đều là \texttt{reranker=True}.
    \item Collection \texttt{semantic} tốt hơn \texttt{fixed\_size} trên cả source hit và evidence hit.
    \item Image exact hit còn thấp; phần lớn chất lượng top-3 đến từ text retrieval.
  \end{itemize}
\end{frame}

\begin{frame}{Kết luận thực nghiệm}
  \begin{itemize}
    \item Pipeline truy hồi đa nhánh kết hợp dense, sparse và visual retrieval đạt hiệu quả cao cho truy hồi bằng chứng văn bản.
    \item Cấu hình nên ưu tiên cho hệ thống hiện tại:
    \[
      \texttt{semantic collection + CLIP + reranker}
    \]
    \item Nếu mục tiêu tối ưu source URL chính xác, Gemini 2.5 Flash là refiner tốt nhất trong thí nghiệm.
    \item Nếu mục tiêu tối ưu evidence-level hit, GPT-4o mini cho kết quả nhỉnh hơn ở \texttt{evidence\_hit\_at\_3}.
  \end{itemize}
\end{frame}
